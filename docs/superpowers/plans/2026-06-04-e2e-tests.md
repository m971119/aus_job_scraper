# E2E Test Suite Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Playwright e2e test suite covering core user flows, runnable locally (backend via Docker) and in GitHub Actions CI (backend via uv).

**Architecture:** `e2e/` at repo root holds its own `package.json` with `@playwright/test`. `global-setup.ts` starts the backend (Docker locally, `uv` in CI), seeds a fresh SQLite test DB via a stdlib-only Python script, then Playwright's `webServer` config starts Next.js dev server. Tests run serially (`workers: 1`) to avoid DB state conflicts.

**Tech Stack:** Playwright, Python sqlite3 (stdlib), Docker Compose, GitHub Actions, Node 22, Python 3.13, uv

```typescript
import { execSync, spawn } from "child_process";
import * as fs from "fs";
import * as path from "path";

const ROOT = path.resolve(__dirname, "..");
const PID_FILE = path.join(__dirname, ".test-pids.json");
const isCI = !!process.env.CI;

async function waitForUrl(url: string, timeoutMs = 60000): Promise<void> {
  const start = Date.now();
  while (Date.now() - start < timeoutMs) {
    try {
      const res = await fetch(url);
      if (res.ok) return;
    } catch {
      // not ready yet
    }
    await new Promise((r) => setTimeout(r, 500));
  }
  throw new Error(`Timed out waiting for ${url}`);
}

export default async function globalSetup() {
  const pids: Record<string, number> = {};

  if (isCI) {
    const backendDir = path.join(ROOT, "backend");
    const backend = spawn("uv", ["run", "uvicorn", "main:app", "--port", "8000"], {
      cwd: backendDir,
      env: { ...process.env, DATABASE_URL: "sqlite:///./data/test.db" },
      detached: true,
      stdio: "ignore",
    });
    backend.unref();
    pids.backend = backend.pid!;
  } else {
    execSync(
      "docker compose -f docker-compose.yml -f docker-compose.test.yml up -d --force-recreate backend",
      { cwd: ROOT, stdio: "inherit" }
    );
  }

  console.log("Waiting for backend health check...");
  await waitForUrl("http://localhost:8000/health");
  console.log("Backend ready.");

  const dbPath = path.join(ROOT, "backend", "data", "test.db");
  execSync(`python3 ${path.join(__dirname, "fixtures", "seed.py")}`, {
    env: { ...process.env, DB_PATH: dbPath },
    stdio: "inherit",
  });

  fs.writeFileSync(PID_FILE, JSON.stringify(pids));
}
```

- [ ] **Step 2: Create `e2e/global-teardown.ts`**

```typescript
import { execSync } from "child_process";
import * as fs from "fs";
import * as path from "path";

const ROOT = path.resolve(__dirname, "..");
const PID_FILE = path.join(__dirname, ".test-pids.json");
const isCI = !!process.env.CI;

export default async function globalTeardown() {
  if (isCI) {
    if (fs.existsSync(PID_FILE)) {
      const pids: Record<string, number> = JSON.parse(
        fs.readFileSync(PID_FILE, "utf8")
      );
      if (pids.backend) {
        try {
          process.kill(pids.backend, "SIGTERM");
        } catch {
          // process may have already exited
        }
      }
      fs.unlinkSync(PID_FILE);
    }
  } else {
    execSync("docker compose stop backend", { cwd: ROOT, stdio: "inherit" });
  }
}
```

import { test, expect } from "@playwright/test";

test.describe("Scrape progress UI", () => {
  test("shows progress indicator after triggering a scrape", async ({ page }) => {
    // Mock POST /api/scrape → 202 accepted
    await page.route("**/api/scrape", (route) => {
      if (route.request().method() === "POST") {
        route.fulfill({ status: 202, json: { status: "started" } });
      } else {
        route.continue();
      }
    });

    // Mock GET /api/scrape/status → always running
    await page.route("**/api/scrape/status", (route) =>
      route.fulfill({
        json: {
          status: "running",
          phase: "seeking",
          current_page: 3,
          jobs_scraped: 12,
          jobs_compared: 0,
          inserted: 0,
          updated_reposts: 0,
          skipped_hidden: 0,
          error: null,
        },
      })
    );

    await page.goto("/");
    await page.waitForLoadState("networkidle");

    // Fill in keywords and submit
    await page.fill('input[placeholder="Keywords (e.g. Python developer)"]', "Python");
    await page.getByRole("button", { name: "Scrape Seek" }).click();

    // Progress text appears
    await expect(page.getByText(/Scraping page/)).toBeVisible();

    // Cancel button appears
    await expect(page.getByRole("button", { name: "Cancel" })).toBeVisible();

    // Submit button becomes disabled with "Scraping..." text
    await expect(page.getByRole("button", { name: "Scraping..." })).toBeDisabled();
  });

  test("cancel button resets the scrape status", async ({ page }) => {
    let statusCallCount = 0;

    await page.route("**/api/scrape", (route) => {
      if (route.request().method() === "POST") {
        route.fulfill({ status: 202, json: { status: "started" } });
      } else {
        route.continue();
      }
    });

    await page.route("**/api/scrape/cancel", (route) =>
      route.fulfill({ json: { status: "cancel_requested" } })
    );

    await page.route("**/api/scrape/status", (route) => {
      statusCallCount++;
      if (statusCallCount >= 3) {
        route.fulfill({
          json: {
            status: "cancelled",
            phase: "seeking",
            current_page: 0,
            jobs_scraped: 0,
            jobs_compared: 0,
            inserted: 0,
            updated_reposts: 0,
            skipped_hidden: 0,
            error: null,
          },
        });
      } else {
        route.fulfill({
          json: {
            status: "running",
            phase: "seeking",
            current_page: 1,
            jobs_scraped: 3,
            jobs_compared: 0,
            inserted: 0,
            updated_reposts: 0,
            skipped_hidden: 0,
            error: null,
          },
        });
      }
    });

    await page.goto("/");
    await page.waitForLoadState("networkidle");

    await page.fill('input[placeholder="Keywords (e.g. Python developer)"]', "Python");
    await page.getByRole("button", { name: "Scrape Seek" }).click();

    // Wait for cancel button
    await page.getByRole("button", { name: "Cancel" }).waitFor();
    await page.getByRole("button", { name: "Cancel" }).click();

    // After cancellation, "Scrape Seek" button becomes available again
    await expect(page.getByRole("button", { name: "Scrape Seek" })).toBeEnabled({ timeout: 10000 });
  });
});
